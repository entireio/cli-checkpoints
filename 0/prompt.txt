```go
func writeExecutableLimited(root *os.Root, destName string, r io.Reader, limit int64) error {
	tmp, tmpName, err := jsonutil.CreateTempIn(root, destName)
	if err != nil {
		return fmt.Errorf("create staging binary: %w", err)
	}
	defer func() { _ = root.Remove(tmpName) }() //nolint:errcheck // no-op once the rename succeeds

	n, err := io.Copy(tmp, io.LimitReader(r, limit+1))
	if err != nil {
		_ = tmp.Close()
		return fmt.Errorf("write binary: %w", err)
	}
	if err := tmp.Close(); err != nil {
		return fmt.Errorf("close binary: %w", err)
	}
	if n &gt; limit {
		return fmt.Errorf("plugin binary exceeds the %d byte limit", limit)
	}
	// CreateTempIn makes 0600; the dispatcher needs it executable.
	if err := root.Chmod(tmpName, 0o755); err != nil {
		return fmt.Errorf("set executable bit: %w", err)
	}
	if err := root.Rename(tmpName, destName); err != nil {
		return fmt.Errorf("place binary: %w", err)
	}
	return nil
}
```

```go
func materializeManagedEntry(root *os.Root, src, destName string, srcInfo os.FileInfo) error {
	// Symlink and Link take src as the target, not as something to resolve
	// inside the root — os.Root refuses to CREATE an absolute symlink but
```

```go
func assetCandidates(name, tag string) []string {
	binName := pluginBinaryPrefix + name
	version := strings.TrimPrefix(tag, "v")
	stems := make([]string, 0, 16)
	for _, osName := range osAliases(runtime.GOOS) {          // darwin -&gt; {darwin, macos}
		for _, arch := range platformArchAliases(runtime.GOOS, runtime.GOARCH) {
			stems = append(stems,
				fmt.Sprintf("%s_%s_%s_%s", binName, version, osName, arch),
				fmt.Sprintf("%s_%s_%s", binName, osName, arch),
			)
		}
	}
	exts := []string{".tar.gz", ".zip", ""}
	if runtime.GOOS == windowsGOOS {
		exts = []string{".zip", ".tar.gz", windowsExeExt}
	}
	var out []string
	for _, ext := range exts {
		for _, stem := range stems {
			out = append(out, stem+ext)
		}
	}
	return out
}
```

Here it did not matter: the release publishes `entire-graph_0.4.0_windows_arm64.zip` and that is what the manifest records.
