```go
func FindMostRecentSession(ctx context.Context) string {
    states, err := ListSessionStates(ctx)
    
    // Scope to current worktree to prevent cross-worktree pollution.
    worktreePath, wpErr := paths.WorktreeRoot(ctx)
    if wpErr == nil &amp;&amp; worktreePath != "" {
        var filtered []*SessionState
        for _, s := range states {
            if s.WorktreePath == worktreePath {  // Filter by current worktree
                filtered = append(filtered, s)
            }
        }
        if len(filtered) &gt; 0 {
            states = filtered  // Use filtered list if matches exist
        }
        // If no sessions match the worktree, fall back to all sessions
    }
    
    // Sort by LastInteractionTime, fall back to StartedAt
    var best *SessionState
    for _, s := range states {
        if s.LastInteractionTime == nil { continue }
        if best == nil || s.LastInteractionTime.After(*best.LastInteractionTime) {
            best = s
        }
    }
    if best != nil { return best.SessionID }
    
    // Fallback: return most recently started session
    for _, s := range states {
        if best == nil || s.StartedAt.After(best.StartedAt) {
            best = s
        }
    }
    if best != nil { return best.SessionID }
    return ""
}
```

## Answers (Cited from Official Documentation)

### 1. **What `prompt_input_exit` means**

From the **hooks.md** documentation, `prompt_input_exit` is triggered when **the user exits the session through the prompt input interface** — specifically keyboard-initiated exits.

### 2. **Does it correspond to keyboard exits (Ctrl-C/Ctrl-D)?**

**Yes, explicitly.** According to the **interactive-mode.md** documentation:

- **Ctrl+C** (when nothing is running): First press clears the prompt input; second press **exits Claude Code**
- **Ctrl+D**: **Exit Claude Code session** (EOF signal)

Both of these keyboard actions trigger `prompt_input_exit` in the SessionEnd hook.

### 3. **Does typing the literal word "exit" or "quit" produce `prompt_input_exit`?**

**No.** The documentation makes clear that:
- Typing "exit" or "quit" as text would be **sent to the model as a normal prompt** — not a session termination
- Only **keyboard signals** (Ctrl+C on the second press, or Ctrl+D) trigger `prompt_input_exit`
- Typing the word "exit" produces no SessionEnd hook at all — it's just conversation text

### 4. **Other `reason` values** (from hooks.md table):

| Reason | Triggered By |
|--------|--------------|
| `clear` | User runs `/clear` command |
| `resume` | User runs `--resume` or `--continue` flag, or `/resume` command |
| `logout` | User logs out |
| `bypass_permissions_disabled` | Bypass permissions mode is disabled |
| `other` | Any other session termination reason |
