  detect git repo

      prompt login

    if not mirrored:

      infer GitHub URL from origin if possible
      if missing URL: ask for GitHub URL
      create/wait for mirror readiness

    entire import claude-code --link-commits --yes --high-confidence

  run matcher

    entire auth login

    entire import claude-code --dry-run --link-commits
