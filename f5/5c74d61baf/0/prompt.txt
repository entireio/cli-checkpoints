- Ref 1: creates a 2-minute deadline (wall-clock, not relative) → could expire while processing refs 2-N
- Ref 2: resets to a new 2-minute deadline (child context, but the parent already has a shorter deadline)

pushCtx, cancel := context.WithTimeout(pushCtx, checkpointPushBudget)  // ONE budget for all recoveries

for _, ref := range existing {

}

func pushCheckpointRefWithRecovery(ctx context.Context, target string, ref plumbing.ReferenceName) error {
